CONSULTA 2: Pantalla de Despacho Comandas en Barra (JOIN múltiple, WHERE, ORDER BY)
SELECT p.id AS comanda_id, m.numero AS mesa, u.nombre_completo AS mesero, pr.nombre AS producto, dp.cantidad, dp.notas_preparacion, p.estado FROM pedido p JOIN mesa m ON m.id = p.mesa_id JOIN usuario u ON u.id = p.usuario_id JOIN detalle_pedido dp ON dp.pedido_id = p.id JOIN producto pr ON pr.id = dp.producto_id WHERE p.estado IN ('PENDIENTE', 'EN_PREPARACION') ORDER BY p.fecha_hora ASC;

CONSULTA 3: MÓDULO INNOVADOR 1 - Ranking Top Vendidos (GROUP BY, SUM, COUNT, ORDER BY)
SELECT pr.codigo, pr.nombre AS producto, c.nombre AS categoria, SUM(dp.cantidad) AS total_unidades_vendidas, COUNT(DISTINCT dp.pedido_id) AS veces_ordenado, SUM(dp.subtotal) AS ingresos_totales_cop FROM detalle_pedido dp JOIN producto pr ON pr.id = dp.producto_id JOIN categoria c ON c.id = pr.categoria_id JOIN pedido p ON p.id = dp.pedido_id WHERE p.estado != 'ANULADO' GROUP BY pr.id, pr.codigo, pr.nombre, c.nombre ORDER BY total_unidades_vendidas DESC;

CONSULTA 4: MÓDULO INNOVADOR 2 - Reporte Costo Financiero de Mermas (GROUP BY, SUM, COUNT)
SELECT me.motivo AS causal_desperdicio, COUNT(me.id) AS cantidad_incidentes, SUM(me.cantidad) AS total_unidades_desechadas, SUM(me.perdida_total) AS costo_financiero_perdida_cop FROM merma me GROUP BY me.motivo ORDER BY costo_financiero_perdida_cop DESC;

CONSULTA 5: Filtrado de Productos Estrella con Ventas Acumuladas > 2 (HAVING, GROUP BY, SUM)
SELECT pr.nombre AS producto, SUM(dp.cantidad) AS unidades_vendidas, SUM(dp.subtotal) AS total_recaudado FROM detalle_pedido dp JOIN producto pr ON pr.id = dp.producto_id GROUP BY pr.id, pr.nombre HAVING SUM(dp.cantidad) > 2 ORDER BY unidades_vendidas DESC;
